cmake_minimum_required(VERSION 3.16)
project(OpenRM95 CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# Dependencies: SDL2, SDL2_mixer and FreeType. CMake config packages (vcpkg, MSYS2, Homebrew ...)
# are preferred, pkg-config is used as a fallback.
find_package(SDL2 CONFIG QUIET)
find_package(SDL2_mixer CONFIG QUIET)
find_package(Freetype QUIET)
if(TARGET SDL2::SDL2 AND TARGET SDL2_mixer::SDL2_mixer AND TARGET Freetype::Freetype)
  set(OPENRM95_LIBS SDL2::SDL2 SDL2_mixer::SDL2_mixer Freetype::Freetype)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(SDL2_MIXER REQUIRED IMPORTED_TARGET SDL2_mixer)
  pkg_check_modules(FREETYPE REQUIRED IMPORTED_TARGET freetype2)
  set(OPENRM95_LIBS PkgConfig::SDL2 PkgConfig::SDL2_MIXER PkgConfig::FREETYPE)
endif()

file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(openrm95 ${SOURCES})
target_include_directories(openrm95 PRIVATE src)
target_compile_definitions(openrm95 PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(openrm95 PRIVATE ${OPENRM95_LIBS})
if(MSVC)
  target_compile_options(openrm95 PRIVATE /W3 /utf-8)
else()
  target_compile_options(openrm95 PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
endif()
install(TARGETS openrm95 RUNTIME DESTINATION bin)
